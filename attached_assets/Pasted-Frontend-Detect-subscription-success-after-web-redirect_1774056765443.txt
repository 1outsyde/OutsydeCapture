Frontend — Detect subscription success after web redirect + show current plan
Since we're using web redirect for Stripe checkout, we need to detect success when the user returns to the app.
Task 1: Add API call for current subscription
In services/api.ts, add:
typescriptexport async function getCurrentSubscription(): Promise<{
  hasSubscription: boolean;
  subscription: {
    id: string;
    tierId: string;
    tierName: string;
    tierDisplayName: string;
    priceInCents: number;
    status: string;
  } | null;
}> {
  const response = await request('/api/subscription/current', { method: 'GET' });
  return response;
}
Task 2: Update SubscriptionPlanScreen with success detection
tsximport { useEffect, useRef, useState } from 'react';
import { AppState, AppStateStatus } from 'react-native';
import { getCurrentSubscription } from '../services/api';

// Add state
const [currentSubscription, setCurrentSubscription] = useState<any>(null);
const [subscribeSuccess, setSubscribeSuccess] = useState(false);
const [checkingSubscription, setCheckingSubscription] = useState(true);
const wasSubscribing = useRef(false);

// Fetch subscription on mount and when app returns to foreground
useEffect(() => {
  const fetchSubscription = async () => {
    try {
      const response = await getCurrentSubscription();
      const newSubscription = response.subscription;
      
      // Detect if user just subscribed (had no subscription before, has one now)
      // OR switched plans (different tier than before)
      if (wasSubscribing.current && newSubscription) {
        if (!currentSubscription || currentSubscription.tierId !== newSubscription.tierId) {
          setSubscribeSuccess(true);
        }
      }
      
      setCurrentSubscription(newSubscription);
      wasSubscribing.current = false;
    } catch (err) {
      console.error('Failed to fetch subscription:', err);
    } finally {
      setCheckingSubscription(false);
    }
  };

  fetchSubscription();

  // Listen for app coming back to foreground (after Stripe redirect)
  const handleAppStateChange = (nextAppState: AppStateStatus) => {
    if (nextAppState === 'active') {
      fetchSubscription();
    }
  };

  const subscription = AppState.addEventListener('change', handleAppStateChange);
  return () => subscription.remove();
}, []);

// Track when user taps subscribe
const handleSubscribe = async () => {
  if (!selectedTier) return;
  
  setSubscribing(true);
  setSubscribeError(null);
  wasSubscribing.current = true; // Mark that we're in subscribe flow
  
  try {
    const response = await createTierSubscriptionCheckout(selectedTier.id);
    if (response.url) {
      await Linking.openURL(response.url);
    }
  } catch (err) {
    setSubscribeError(toErrorString(err));
    wasSubscribing.current = false;
  } finally {
    setSubscribing(false);
  }
};
